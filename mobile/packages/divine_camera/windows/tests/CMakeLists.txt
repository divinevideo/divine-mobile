cmake_minimum_required(VERSION 3.19)
project(divine_capture_ownership LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(capture_ownership_test ../../native/tests/capture_ownership_test.cc)
target_compile_features(capture_ownership_test PRIVATE cxx_std_17)
target_include_directories(capture_ownership_test PRIVATE ../../native/include)
target_link_libraries(capture_ownership_test PRIVATE Threads::Threads)
foreach(capture_platform linux windows)
  add_executable(${capture_platform}_pixel_utils_test ../../native/tests/${capture_platform}_pixel_utils_test.cc)
  target_compile_features(${capture_platform}_pixel_utils_test PRIVATE cxx_std_17)
  target_include_directories(${capture_platform}_pixel_utils_test PRIVATE ../../${capture_platform}/capture)
endforeach()
add_executable(capture_initialization_test ../../native/tests/capture_initialization_test.cc)
target_compile_features(capture_initialization_test PRIVATE cxx_std_17)
target_include_directories(capture_initialization_test PRIVATE ../../native/include)
target_link_libraries(capture_initialization_test PRIVATE Threads::Threads)
add_executable(pixel_transform_test ../../native/tests/pixel_transform_test.cc)
target_compile_features(pixel_transform_test PRIVATE cxx_std_17)
target_include_directories(pixel_transform_test PRIVATE ../../native/include)
include(CTest)
add_test(NAME capture_output_and_writer_release COMMAND capture_ownership_test)
add_test(NAME capture_initialization_terminal_ownership COMMAND capture_initialization_test)
add_test(NAME saved_output_pixel_mirroring COMMAND pixel_transform_test)
add_test(NAME linux_pixel_conversion COMMAND linux_pixel_utils_test)
add_test(NAME windows_pixel_conversion COMMAND windows_pixel_utils_test)
set_tests_properties(capture_output_and_writer_release PROPERTIES TIMEOUT 30)

if(WIN32)
  add_executable(capture_trace_launcher capture_trace_launcher.cc)
  target_compile_features(capture_trace_launcher PRIVATE cxx_std_17)
  target_compile_definitions(capture_trace_launcher PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  set_target_properties(capture_trace_launcher PROPERTIES OUTPUT_NAME "capture trace launcher")
  add_test(NAME capture_trace_process_tree_release COMMAND capture_trace_launcher --self-test)
  set_tests_properties(capture_trace_process_tree_release PROPERTIES TIMEOUT 20)
  add_executable(synthetic_audio_source_test
    synthetic_audio_source_test.cc ../capture/fake_camera_source.cpp)
  target_compile_features(synthetic_audio_source_test PRIVATE cxx_std_17)
  target_compile_definitions(synthetic_audio_source_test PRIVATE CAMERA_DESKTOP_FAKE_CAMERA)
  target_include_directories(synthetic_audio_source_test PRIVATE ../capture ../../native/include)
  target_link_libraries(synthetic_audio_source_test PRIVATE mfreadwrite mfplat mfuuid ole32 propsys)
  add_test(NAME synthetic_audio_source_contract COMMAND synthetic_audio_source_test)
  set_tests_properties(synthetic_audio_source_contract PROPERTIES TIMEOUT 30)
endif()
