cmake_minimum_required(VERSION 3.16)
project(divine_capture_tests LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CAPTURE_GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0 gio-2.0)
add_executable(recording_probe tests/recording_probe.cc ../linux/capture/record_handler.cc ../linux/capture/photo_handler.cc)
target_compile_features(recording_probe PRIVATE cxx_std_17)
target_compile_definitions(recording_probe PRIVATE DIVINE_CAMERA_TEST_CAPTURE)
target_include_directories(recording_probe PRIVATE ../linux/capture)
target_link_libraries(recording_probe PRIVATE PkgConfig::CAPTURE_GST)
add_executable(portal_concurrency_probe tests/portal_concurrency_probe.cc ../linux/capture/pipewire_portal.cc)
target_compile_features(portal_concurrency_probe PRIVATE cxx_std_17)
target_include_directories(portal_concurrency_probe PRIVATE ../linux/capture)
target_link_libraries(portal_concurrency_probe PRIVATE PkgConfig::CAPTURE_GST)
find_package(Threads REQUIRED)
add_executable(capture_ownership_test tests/capture_ownership_test.cc)
target_compile_features(capture_ownership_test PRIVATE cxx_std_17)
target_include_directories(capture_ownership_test PRIVATE include)
target_link_libraries(capture_ownership_test PRIVATE Threads::Threads)
foreach(capture_platform linux windows)
  add_executable(${capture_platform}_pixel_utils_test tests/${capture_platform}_pixel_utils_test.cc)
  target_compile_features(${capture_platform}_pixel_utils_test PRIVATE cxx_std_17)
  target_include_directories(${capture_platform}_pixel_utils_test PRIVATE ../${capture_platform}/capture)
endforeach()
option(DIVINE_CAPTURE_TEST_ASAN "Instrument the native recording test" OFF)
if(DIVINE_CAPTURE_TEST_ASAN)
  foreach(capture_test recording_probe portal_concurrency_probe capture_ownership_test linux_pixel_utils_test windows_pixel_utils_test)
    target_compile_options(${capture_test} PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
    target_link_options(${capture_test} PRIVATE -fsanitize=address)
  endforeach()
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME concurrent_portal_enumeration COMMAND portal_concurrency_probe)
  add_test(NAME pending_portal_disposal COMMAND portal_concurrency_probe dispose)
  set_tests_properties(concurrent_portal_enumeration pending_portal_disposal
    PROPERTIES TIMEOUT 15 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
  add_test(NAME capture_output_and_writer_release COMMAND capture_ownership_test)
  add_test(NAME linux_pixel_conversion COMMAND linux_pixel_utils_test)
  add_test(NAME windows_pixel_conversion COMMAND windows_pixel_utils_test)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/recording-output")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/audio-recovery-output")
  add_test(NAME repeated_av_capture
    COMMAND recording_probe "${CMAKE_CURRENT_BINARY_DIR}/recording-output")
  add_test(NAME microphone_setup_failure_and_retry
    COMMAND recording_probe "${CMAKE_CURRENT_BINARY_DIR}/audio-recovery-output" audio-unavailable)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/audio-start-failure-output")
  add_test(NAME microphone_start_failure_releases_writer
    COMMAND recording_probe "${CMAKE_CURRENT_BINARY_DIR}/audio-start-failure-output" audio-start-failure)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/production-audio-failure-output")
  add_test(NAME production_microphone_unavailable
    COMMAND recording_probe "${CMAKE_CURRENT_BINARY_DIR}/production-audio-failure-output" no-device-audio)
  set_tests_properties(production_microphone_unavailable PROPERTIES TIMEOUT 30
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;GST_PLUGIN_FEATURE_RANK=openalsrc:0,pulsesrc:0,alsasrc:0,oss4src:0,osssrc:0,pipewiresrc:0,jackaudiosrc:0")
  set_tests_properties(repeated_av_capture microphone_setup_failure_and_retry microphone_start_failure_releases_writer
    PROPERTIES TIMEOUT 45 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;G_DEBUG=fatal-criticals")
endif()
