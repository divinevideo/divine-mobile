cmake_minimum_required(VERSION 3.19)
project(divine_worker_lifetime_tests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows worker lifetime tests require Windows.")
endif()
add_executable(worker_lifetime_test worker_lifetime_test.cpp ../worker_lifetime.cpp)
target_compile_features(worker_lifetime_test PRIVATE cxx_std_17)
target_compile_definitions(worker_lifetime_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(worker_lifetime_test PRIVATE /W4 /WX)
enable_testing()
add_test(NAME owned_worker_lifetime COMMAND worker_lifetime_test)
set_tests_properties(owned_worker_lifetime PROPERTIES TIMEOUT 30)
