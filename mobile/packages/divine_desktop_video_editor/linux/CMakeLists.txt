cmake_minimum_required(VERSION 3.19)
project(divine_desktop_video_editor LANGUAGES CXX)
set(PLUGIN_NAME "divine_desktop_video_editor_plugin")
add_library(${PLUGIN_NAME} SHARED divine_desktop_video_editor_plugin.cc)
apply_standard_settings(${PLUGIN_NAME})
target_compile_features(${PLUGIN_NAME} PRIVATE cxx_std_17)
set_target_properties(${PLUGIN_NAME} PROPERTIES CXX_VISIBILITY_PRESET hidden)
target_compile_definitions(${PLUGIN_NAME} PRIVATE FLUTTER_PLUGIN_IMPL)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter PkgConfig::GTK)
add_subdirectory("../native" "${CMAKE_CURRENT_BINARY_DIR}/media")
add_dependencies(${PLUGIN_NAME} divine_media_worker)
set(divine_desktop_video_editor_bundled_libraries "" PARENT_SCOPE)

# Flutter installs its bundle by first removing the destination. Schedule this
# after the application's install rules so that cleanup cannot erase the helper.
function(_divine_install_media_worker)
  install(TARGETS divine_media_worker RUNTIME DESTINATION "lib/divine_media" COMPONENT Runtime)
endfunction()
cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL _divine_install_media_worker)
