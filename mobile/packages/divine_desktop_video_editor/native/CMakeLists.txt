cmake_minimum_required(VERSION 3.16)
project(divine_media_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(divine_media_worker media_worker.cc media_common.cc media_renderer.cc media_reader.cc)
target_link_libraries(divine_media_worker PRIVATE Threads::Threads)
if(WIN32)
  set(DIVINE_GSTREAMER_ROOT "$ENV{GSTREAMER_1_0_ROOT_MSVC_X86_64}" CACHE PATH "Private GStreamer development SDK")
  if(NOT EXISTS "${DIVINE_GSTREAMER_ROOT}/include/gstreamer-1.0/ges/ges.h")
    message(FATAL_ERROR "Set DIVINE_GSTREAMER_ROOT or GSTREAMER_1_0_ROOT_MSVC_X86_64 to the GStreamer MSVC x64 development SDK.")
  endif()
  target_include_directories(divine_media_worker PRIVATE
    "${DIVINE_GSTREAMER_ROOT}/include/gstreamer-1.0"
    "${DIVINE_GSTREAMER_ROOT}/include/glib-2.0"
    "${DIVINE_GSTREAMER_ROOT}/lib/glib-2.0/include"
    "${DIVINE_GSTREAMER_ROOT}/include/json-glib-1.0")
  foreach(component gstapp-1.0 gstvideo-1.0 ges-1.0 gstpbutils-1.0 gstreamer-1.0 json-glib-1.0 gobject-2.0 glib-2.0)
    string(REPLACE "-" "_" key "${component}")
    string(REPLACE "." "_" key "${key}")
    find_library(DIVINE_${key} NAMES "${component}" "lib${component}"
      PATHS "${DIVINE_GSTREAMER_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(divine_media_worker PRIVATE "${DIVINE_${key}}")
  endforeach()
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GES REQUIRED IMPORTED_TARGET gst-editing-services-1.0)
  pkg_check_modules(READERS REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0)
  pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-glib-1.0)
  target_link_libraries(divine_media_worker PRIVATE PkgConfig::GES PkgConfig::JSON PkgConfig::READERS)
endif()
if(MSVC)
  target_compile_definitions(divine_media_worker PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(divine_media_worker PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(divine_media_worker PRIVATE -Wall -Wextra -Werror)
endif()
