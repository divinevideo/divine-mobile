import 'dart:convert';
import 'dart:io';

import 'package:db_client/db_client.dart';
import 'package:dm_repository/dm_repository.dart';
import 'package:drift/native.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:follow_repository/follow_repository.dart';
import 'package:http/http.dart' as http;
import 'package:mocktail/mocktail.dart';
import 'package:models/models.dart';
import 'package:nostr_client/nostr_client.dart';
import 'package:openvine/services/user_ranking/user_ranking_signals_loader.dart';
import 'package:profile_repository/profile_repository.dart';

class _Dm extends Mock implements DmRepository {}

class _Follow extends Mock implements FollowRepository {}

class _Nostr extends Mock implements NostrClient {}

String keyOf(int i) => i.toRadixString(16).padLeft(64, '0');
void main() {
  group('manual host benchmark', () {
    test('real SQLite loader and cached name read', () async {
      final directory = await Directory.systemTemp.createTemp(
        'ranking-benchmark-',
      );
      final file = File('${directory.path}/ranking.sqlite');
      final seed = AppDatabase.test(NativeDatabase(file));
      final now = DateTime.now().toUtc();
      final seconds = now.millisecondsSinceEpoch ~/ 1000;
      final owner = keyOf(10001);
      final keys = [for (var i = 1; i <= 5000; i++) keyOf(i)];
      await seed.transaction(() async {
        await seed.userProfilesDao.upsertProfiles([
          for (var i = 0; i < keys.length; i++)
            UserProfile(
              pubkey: keys[i],
              displayName: 'Person $i',
              name: 'person$i',
              rawData: const {},
              createdAt: now,
              eventId: keyOf(20000 + i),
            ),
        ]);
        for (var i = 0; i < keys.length; i++) {
          final key = keys[i];
          await seed
              .into(seed.profileStats)
              .insert(
                ProfileStatRow(
                  pubkey: key,
                  cachedAt: now,
                  videoCount: i % 2000,
                ),
              );
          if (i.isEven)
            await seed.customStatement(
              'INSERT INTO my_followers(owner_pubkey,follower_pubkey,last_seen_at) VALUES (?,?,?)',
              [owner, key, seconds],
            );
          await seed.customStatement(
            'INSERT INTO event(id,pubkey,created_at,kind,tags,content,sig) VALUES (?,?,?,?,?,?,?)',
            [
              keyOf(30000 + i),
              key,
              seconds - i * 60,
              34236,
              '[]',
              '',
              '0' * 128,
            ],
          );
          if (i % 7 == 0)
            await seed.customStatement(
              'INSERT INTO nip05_verifications(pubkey,nip05,status,verified_at,expires_at) VALUES (?,?,?,?,?)',
              [
                key,
                'person$i@example.invalid',
                'verified',
                seconds,
                seconds + 86400,
              ],
            );
          if (i % 5 == 0)
            await seed.conversationsDao.upsertConversation(
              id: key,
              participantPubkeys: jsonEncode([owner, key]),
              isGroup: false,
              createdAt: seconds - 86400,
              lastMessageTimestamp: seconds - i,
              currentUserHasSent: true,
              ownerPubkey: owner,
            );
        }
      });
      await seed.close();
      final db = AppDatabase.test(NativeDatabase(file));
      final client = http.Client();
      final profiles = ProfileRepository(
        nostrClient: _Nostr(),
        userProfilesDao: db.userProfilesDao,
        httpClient: client,
      );
      final dms = _Dm();
      final follows = _Follow();
      when(() => follows.followingPubkeys).thenReturn(keys);
      when(() => dms.watchAcceptedConversations()).thenAnswer(
        (_) => db.conversationsDao
            .watchAcceptedConversations(ownerPubkey: owner)
            .map(
              (rows) => [
                for (final r in rows)
                  DmConversation(
                    id: r.id,
                    participantPubkeys: (jsonDecode(
                      r.participantPubkeys,
                    ) as List).cast<String>(),
                    isGroup: r.isGroup,
                    createdAt: r.createdAt,
                    lastMessageTimestamp: r.lastMessageTimestamp,
                    currentUserHasSent: r.currentUserHasSent,
                  ),
              ],
            ),
      );
      final loader = UserRankingSignalsLoader(
        dmRepository: dms,
        followRepository: follows,
        profileRepository: profiles,
        myFollowersDao: db.myFollowersDao,
        profileStatsDao: db.profileStatsDao,
        nostrEventsDao: db.nostrEventsDao,
        nip05VerificationsDao: db.nip05VerificationsDao,
        currentUserPubkey: owner,
      );
      final samples = <int>[];
      final loadSamples = <int>[];
      final nameSamples = <int>[];
      for (var run = 0; run < 11; run++) {
        final watch = Stopwatch()..start();
        final signals = await loader.load(keys);
        final loadedAt = watch.elapsedMicroseconds;
        final names = await loader.loadCachedDisplayNames(keys);
        watch.stop();
        expect(signals.length, 5000);
        expect(names.length, 5000);
        expect(signals[keys.first]!.videoCount, 0);
        samples.add(watch.elapsedMicroseconds);
        loadSamples.add(loadedAt);
        nameSamples.add(watch.elapsedMicroseconds - loadedAt);
      }
      final warm = samples.skip(1).toList()..sort();
      final loads = loadSamples.skip(1).toList()..sort();
      final namesTimes = nameSamples.skip(1).toList()..sort();
      stdout.writeln(
        'Warm stage medians: loader.load ${(loads[4] + loads[5]) / 2000} ms; names ${(namesTimes[4] + namesTimes[5]) / 2000} ms (separately computed medians).',
      );
      final indexes = await db
          .customSelect(
            "SELECT name, sql FROM sqlite_master WHERE type='index' ORDER BY name",
          )
          .get();
      stdout.writeln('SQLite indexes: ${indexes.map((r) => r.data).toList()}');
      stdout.writeln(
        'HOST REAL SQLITE: 5000 profiles/stats/videos/follows; 2500 followers, 1000 accepted DMs, 715 verifications.',
      );
      stdout.writeln(
        'Full loader.load + loadCachedDisplayNames; reopen-first ${samples.first / 1000} ms; warm 10 median ${(warm[4] + warm[5]) / 2000} ms min ${warm.first / 1000} max ${warm.last / 1000}.',
      );
      stdout.writeln(
        'Flutter-test host JIT, not profile-device; seeded OS caches; mock follow repository returns a list; production loader snapshots its Set inside each measured load; DM adapter maps real DAO rows, no repository decryption overhead. Does NOT verify 50 ms phone budget.',
      );
      await db.close();
      client.close();
      await directory.delete(recursive: true);
    });
  });
}
