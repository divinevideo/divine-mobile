import 'helpers/test_provider_overrides.dart';

class _Loader extends Mock implements UserRankingSignalsLoader {}

class _Share extends Mock implements VideoSharingService {}

class _Session extends NostrSession {
  @override
  NostrSessionReadiness build() =>
      NostrSessionReadiness.identityKnown(pubkey: owner);
}

const owner =
    'ffffffffffffffffffffffffffffffffffffffffffffffffffffffffffffffff';
String keyOf(int i) => i.toRadixString(16).padLeft(64, '0');
void main() {
  group('documentation capture', () {
    for (final dark in [true, false]) {
      for (final surface in ['share-sheet', 'dm-picker', 'following']) {
        testWidgets('$surface ${dark ? 'dark' : 'light'}', (tester) async {
          tester.view.physicalSize = const Size(390, 844);
          tester.view.devicePixelRatio = 1;
          addTearDown(tester.view.resetPhysicalSize);
          addTearDown(tester.view.resetDevicePixelRatio);
          for (final channel in [
            'pro_video_editor_progress',
            'pro_video_editor_logs',
          ]) {
            tester.binding.defaultBinaryMessenger.setMockMethodCallHandler(
              MethodChannel(channel),
              (_) async => null,
            );
            addTearDown(
              () => tester.binding.defaultBinaryMessenger
                  .setMockMethodCallHandler(MethodChannel(channel), null),
            );
          }
          final profiles = createMockProfileRepository();
          when(profiles.watchVanishedPubkeys)
              .thenAnswer((_) => const Stream<Set<String>>.empty());
          final follows = createMockFollowRepository();
          final loader = _Loader();
          final share = _Share();
          final names = [
            'Mira Loops',
            'Jules Motion',
            'Casey Frames',
            'Rowan Makes',
            'Sage Sketches',
          ];
          final people = [
            for (var i = 0; i < names.length; i++)
              UserProfile(
                pubkey: keyOf(i + 1),
                displayName: names[i],
                name: 'person${i + 1}',
                rawData: const {},
                createdAt: DateTime(2026),
                eventId: keyOf(100 + i),
              ),
          ];
          final keys = people.map((p) => p.pubkey).toList();
          when(() => follows.followingPubkeys).thenReturn(keys);
          when(() => follows.followingStream)
              .thenAnswer((_) => Stream.value(keys));
          when(() => follows.watchMyFollowingCached()).thenAnswer(
            (_) => Stream.value(
              CacheResult.live(
                FollowingSnapshot(pubkeys: keys, count: keys.length),
              ),
            ),
          );
          when(() => follows.isFollowing(any())).thenReturn(true);
          when(() => profiles.getCachedProfiles(pubkeys: any(named: 'pubkeys')))
              .thenAnswer((_) async => people);
          when(
            () => profiles.fetchBatchProfiles(pubkeys: any(named: 'pubkeys')),
          ).thenAnswer((_) async => {for (final p in people) p.pubkey: p});
          for (final person in people) {
            when(() => profiles.getCachedProfile(pubkey: person.pubkey))
                .thenAnswer((_) async => person);
          }
          when(() => loader.isEligible(any()))
              .thenAnswer((i) => i.positionalArguments.first != owner);
          final signals = {
            for (var i = 0; i < people.length; i++)
              people[i].pubkey: UserRankingSignals(
                pubkey: people[i].pubkey,
                youFollow: true,
                followsYou: i == 0,
                lastInteractionAt: i == 0 ? clock.now() : null,
                videoCount: 18 + i * 11,
                lastPostedAt: clock.now().subtract(Duration(hours: 2 + i)),
              ),
          };
          when(() => loader.load(any())).thenAnswer((_) async => signals);
          when(() => loader.load(any(), includeDmCounterparts: true))
              .thenAnswer((_) async => signals);
          when(() => loader.loadCachedDisplayNames(any())).thenAnswer(
            (_) async => {for (final p in people) p.pubkey: p.bestDisplayName},
          );
          when(() => share.recentlySharedWith).thenReturn([]);
          final video = VideoEvent(
            id: keyOf(101),
            pubkey: keyOf(102),
            createdAt: 1757385263,
            content: 'A little loop',
            timestamp: DateTime(2026),
            title: 'A little loop',
          );
          final boundary = GlobalKey();
          await tester.pumpWidget(
            RepaintBoundary(
              key: boundary,
              child: testProviderScope(
                mockAuthService: createMockAuthService(),
                mockProfileRepository: profiles,
                mockFollowRepository: follows,
                rankingSignalsLoader: loader,
                additionalOverrides: [
                  if (surface != 'share-sheet')
                    nostrSessionProvider.overrideWith(_Session.new),
                  videoSharingServiceProvider.overrideWith((ref) => share),
                  for (final person in people)
                    userProfileReactiveProvider(person.pubkey)
                        .overrideWith((ref) => Stream.value(person)),
                ],
                child: MaterialApp(
                  debugShowCheckedModeBanner: false,
                  theme: dark ? VineTheme.theme : VineTheme.lightTheme,
                  localizationsDelegates: appLocalizationsDelegates,
                  supportedLocales: AppLocalizations.supportedLocales,
                  home: switch (surface) {
                    'following' => const MyFollowingScreen(displayName: 'Alex'),
                    'dm-picker' => Scaffold(
                      body: const NewMessageSheet(
                        currentUserPubkey: owner,
                        allowGroups: true,
                      ),
                    ),
                    _ => Scaffold(
                      body: Center(child: ShareActionButton(video: video)),
                    ),
                  },
                ),
              ),
            ),
          );
          await tester.pump();
          await tester.pump(const Duration(milliseconds: 500));
          if (surface == 'share-sheet') {
            await tester.tap(find.byType(ShareActionButton));
            await tester.pump();
            await tester.pump(const Duration(milliseconds: 500));
          }
          for (var frame = 0; frame < 20; frame++) {
            await tester.pump(const Duration(milliseconds: 50));
          }
          await tester.runAsync(GoogleFonts.pendingFonts);
          expect(find.textContaining('Mira'), findsWidgets);
          await tester.pumpAndSettle();
          await tester.pump();
          final render =
              boundary.currentContext!.findRenderObject()!
                  as RenderRepaintBoundary;
          await tester.runAsync(() async {
            final image = await render.toImage(pixelRatio: 2);
            final data = await image.toByteData(format: ui.ImageByteFormat.png);
            final file = File(
              'docs/screenshots/smart-user-ranking/$surface-${dark ? 'dark' : 'light'}.png',
            );
            await file.parent.create(recursive: true);
            await file.writeAsBytes(data!.buffer.asUint8List());
            image.dispose();
          });
          await tester.pumpWidget(const SizedBox());
          await tester.pump();
          await tester.pump(const Duration(milliseconds: 500));
        });
      }
    }
  });
}
